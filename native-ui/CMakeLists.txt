cmake_minimum_required(VERSION 3.20)
project(LocalAudioNativeUi LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use MSVC x64 for the Qt ABI supplied by TS3 3.6.2")
endif()
find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Core Gui Widgets)
add_library(ningq_denoise_ui SHARED ../audio/src/denoise_qt_ui.cpp)
set_target_properties(ningq_denoise_ui PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
target_compile_definitions(ningq_denoise_ui PRIVATE UNICODE _UNICODE NOMINMAX QT_NO_DEBUG)
get_filename_component(LA_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
target_compile_options(ningq_denoise_ui PRIVATE /utf-8 /O2 /experimental:deterministic /pathmap:${LA_PROJECT_ROOT}=.)
target_link_libraries(ningq_denoise_ui PRIVATE Qt5::Core Qt5::Gui Qt5::Widgets user32)
