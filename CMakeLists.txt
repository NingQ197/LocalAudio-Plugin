cmake_minimum_required(VERSION 3.20)
project(LocalAudio VERSION 2.5.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LA_EDITION audio CACHE STRING "audio or screen")
set_property(CACHE LA_EDITION PROPERTY STRINGS audio screen)
if(NOT LA_EDITION MATCHES "^(audio|screen)$")
  message(FATAL_ERROR "LA_EDITION must be audio or screen")
endif()
set(LA_RESOURCE_BUNDLE "" CACHE PATH "Generated by tools/prepare_resources.py")
foreach(header release_assets_manifest.h release_denoise_manifest.h)
  if(NOT EXISTS "${LA_RESOURCE_BUNDLE}/include/${header}")
    message(FATAL_ERROR "Prepare the per-build resource bundle first")
  endif()
endforeach()
foreach(path third_party/rnnoise/include/rnnoise.h third_party/json/json.hpp third_party/webview2/sdk/build/native/include/WebView2.h)
  if(NOT EXISTS "${CMAKE_SOURCE_DIR}/${path}")
    message(FATAL_ERROR "Run python tools/fetch_dependencies.py first: missing ${path}")
  endif()
endforeach()
set(RN_NAMES denoise rnn pitch kiss_fft celt_lpc nnet nnet_default parse_lpcnet_weights rnnoise_data rnnoise_tables)
set(RN_FILES)
foreach(name IN LISTS RN_NAMES)
  list(APPEND RN_FILES "third_party/rnnoise/src/${name}.c")
endforeach()
add_library(rnnoise STATIC ${RN_FILES})
target_include_directories(rnnoise PRIVATE third_party/rnnoise/include third_party/rnnoise/src)
target_compile_definitions(rnnoise PRIVATE RNNOISE_BUILD NDEBUG)
file(GLOB CORE CONFIGURE_DEPENDS "${LA_EDITION}/src/*.cpp")
list(FILTER CORE EXCLUDE REGEX "denoise_qt_ui\\.cpp$")
add_library(ts3localaudio SHARED ${CORE})
target_include_directories(ts3localaudio PRIVATE "${LA_EDITION}/src" third_party/ts3sdk "${LA_RESOURCE_BUNDLE}/include")
target_compile_definitions(ts3localaudio PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(ts3localaudio PRIVATE rnnoise ole32 oleaut32 uuid mmdevapi comctl32 shell32 version user32 gdi32 gdiplus crypt32 bcrypt shlwapi)
if(LA_EDITION STREQUAL "screen")
  target_link_libraries(ts3localaudio PRIVATE dnsapi)
endif()
set_target_properties(ts3localaudio PROPERTIES PREFIX "" OUTPUT_NAME ts3localaudio)
add_executable(client-check installer/client_check.cpp)
target_compile_definitions(client-check PRIVATE UNICODE _UNICODE)
if(MSVC)
  target_compile_options(ts3localaudio PRIVATE /utf-8 /O2 /experimental:deterministic /pathmap:${CMAKE_SOURCE_DIR}=.)
  set_property(TARGET ts3localaudio rnnoise client-check PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
else()
  target_compile_options(ts3localaudio PRIVATE -O2 "-ffile-prefix-map=${CMAKE_SOURCE_DIR}=." "-fmacro-prefix-map=${CMAKE_SOURCE_DIR}=.")
  target_link_options(ts3localaudio PRIVATE -static -Wl,--dynamicbase,--nxcompat,--high-entropy-va)
  target_link_options(client-check PRIVATE -static -municode)
endif()
